# 数据结构之模拟堆
```c++
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N=1e5;
//ph:插入顺序对应的下标
//hp:下标对应的插入顺序
ll h[N],hp[N],ph[N],siz,n;
void swap_heap(int a,int b)
{
    swap(ph[hp[a]],ph[hp[b]]);
    swap(hp[a],hp[b]);
    swap(h[a],h[b]);
}
void down(int x)
{
    int t=x;
    if(2*x <=siz && h[2*x]<h[t]) t=2*x;
    if(2*x +1<=siz && h[2*x+1]<h[t]) t=2*x+1;
    if(t!=x)
    {
        swap_heap(t,x);
        down(t);
    }
}
void up(int x)
{
    while(x/2 && h[x]<h[x/2])
    {
        swap_heap(x,x/2);
        x/=2;
    }
    
}
void insert(int x)
{
    n++;
    siz++;
    ph[n]=siz;
    hp[siz]=n;
    h[siz]=x;
    
}
int main()
{
    int m;
    cin>>m;
    while(m--)
    {
        string s;
        int k,x;
        cin>>s;
        if(s=="I")
        {
            cin>>k;
            insert(k);
            up(siz),down(siz);
        }
        if(s=="PM") cout<<h[1]<<"\n";
        if(s=="DM")
        {
            swap_heap(1,siz);
            siz--;
            down(1);
        }
        if(s=="D")
        {
            cin>>k;
            int pos=ph[k];
            swap_heap(pos,siz);
            siz--;
            up(pos),down(pos);
        }
        if(s=="C")
        {
            cin>>k>>x;
            h[ph[k]]=x;
            up(ph[k]),down(ph[k]);
        }
    }
    return 0;
}
```
*** 
## 注意事项：
- 模拟堆和排序最大的不同就是模拟堆多了两个条件
- 删除第k个插入的数 修改第k个插入的数
- 看似是两个简单的要求，但是我们的堆时不断变化的，我们要把插入的数的次序实时更新它的下标
- 那么我们就得创建两个数组ph和hp它们两个来相互对应 实现实时更新
- 理解了这两个数组的意义后，后面的操作只要细心就行了
- 比如在删除第一个元素时,我们不应该直接赋值，要注意到这里的下标和插入顺序，所以我们选择交换第一个数和最后一个数的位置
- 这样第一个数的坐标就变成了size了，我们再size--就实现删去了，最后再down（1）就行了
- 比如我们在删去第k个插入的数时，我们首先要拿一个pos来记录ph[k]的位置，因为交换后ph[k]也随着改变了，那我们后面的down和up操作就没意义了
- 再就是我们up函数里面，因为我们下标是从一开始的，所以while（x）没什么意义，于是我们用while（x/2）来判断是否到达了第一个位置，后x/=2，继续向上比较
