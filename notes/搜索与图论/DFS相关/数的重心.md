# DFS中的数的重心问题
```c++
#include <bits/stdc++.h>
using namespace std;
const int N=1e5+10;
int h[2*N],e[2*N],ne[2*N],idx;
int ans=N,n;
bool st[2*N];
//新建联系 （a->b）
void add(int a,int b)
{
    e[idx]=b;
    ne[idx]=h[a];
    h[a]=idx++;
}
int dfs(int u)
{
    int sum=1,res=0;
    st[u]=true;
    for(int i=h[u];i!=-1;i=ne[i])
    {
        int j=e[i];
        if(!st[j])
        {
            int s=dfs(j);
            res=max(s,res);
            sum+=s;
        }
    }
    res=max(res,n-sum);
    ans=min(res,ans);
    return sum;
}
int main()
{
    cin>>n;
    int m=n-1;
    memset(h,-1,sizeof h);
    while(m--)
    {
        int a,b;
        cin>>a>>b;
        add(a,b);
        add(b,a);
    }
    dfs(1);
    cout<<ans;
    return 0;
}
```
## 注意事项：
- 这里之所以选择dfs是因为从某一个端点直接往下走了，这一路上顺便把每个节点的都算出来了
- 那剩下节点怎么处理呢，res=max(res,n-sum)就巧妙的处理了没有遍历到的
- 还有几个易错点
- 在递归调用时每到一个新的点都要把它的转态变为true,注意它的状态不用还原，因为只要走一次
- 还有就是n是定义为全局变量的，所以在主函数中循环导入a,b的值时，用m代替n
